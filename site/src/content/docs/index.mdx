import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid>
  <Card title="Portal rendering" icon="document">
    `<Frame>` and `useIframe` render React children into an iframe, and mount only after
    the iframe's own document has loaded, not into the temporary `about:blank` before it.
  </Card>
  <Card title="Auto-resize" icon="seti:image">
    `useIframeResize` follows the content's size, same-origin or cross-origin, with a guard
    against content that grows with the iframe forever.
  </Card>
  <Card title="Typed RPC and events" icon="rocket">
    Describe each side once with `Side<>`, then call methods and emit events in both
    directions. Calls made before the handshake is complete are queued.
  </Card>
  <Card title="Safe by default" icon="approve-check">
    Origins are checked on both sides, and the child must list the parents it accepts.
    `'*'` needs an explicit `unsafeAllowAnyOrigin`.
  </Card>
</CardGrid>
